cmake_minimum_required(VERSION 3.16)
project(ego_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(pcl_ros REQUIRED)
find_package(legbot_runtime REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Bspline.msg" "msg/DataDisp.msg" DEPENDENCIES std_msgs geometry_msgs builtin_interfaces)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
include_directories(algorithm/plan_manage/include algorithm/plan_env/include algorithm/path_searching/include algorithm/bspline_opt/include algorithm/traj_utils/include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_executable(ego_planner_node algorithm/plan_manage/src/ego_planner_node.cpp algorithm/plan_manage/src/ego_replan_fsm.cpp algorithm/plan_manage/src/planner_manager.cpp algorithm/plan_env/src/grid_map.cpp algorithm/plan_env/src/raycast.cpp algorithm/path_searching/src/dyn_a_star.cpp algorithm/bspline_opt/src/uniform_bspline.cpp algorithm/bspline_opt/src/bspline_optimizer.cpp algorithm/bspline_opt/src/gradient_descent_optimizer.cpp algorithm/traj_utils/src/planning_visualization.cpp algorithm/traj_utils/src/polynomial_traj.cpp)
ament_target_dependencies(ego_planner_node rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs tf2 tf2_ros tf2_geometry_msgs message_filters cv_bridge pcl_conversions pcl_ros legbot_runtime)
target_link_libraries(ego_planner_node "${typesupport}" ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
add_executable(traj_server algorithm/plan_manage/src/traj_server.cpp algorithm/bspline_opt/src/uniform_bspline.cpp)
ament_target_dependencies(traj_server rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs tf2 tf2_ros tf2_geometry_msgs message_filters cv_bridge pcl_conversions pcl_ros legbot_runtime)
target_link_libraries(traj_server "${typesupport}" ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
install(TARGETS ego_planner_node traj_server DESTINATION lib/${PROJECT_NAME})
ament_package()
